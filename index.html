-- ============================================ -- أليكس شوب — خطوة قاعدة البيانات للخصائص الجديدة -- (بيانات المندوب + أكواد الخصم) -- الصق الكود ده كله في Supabase > SQL Editor ودوس Run -- ============================================ -- 1) أعمدة جديدة في جدول الطلبات: المندوب + الخصم alter table public.orders add column if not exists driver_name text, add column if not exists driver_phone text, add column if not exists discount numeric not null default 0, add column if not exists discount_code text; -- 2) جدول أكواد الخصم (الأدمن بس اللي يشوفه ويعدّل فيه) create table if not exists public.discount_codes ( code text primary key, value numeric not null check (value > 0), active boolean not null default true, created_at timestamptz not null default now() ); alter table public.discount_codes enable row level security; drop policy if exists "admin manages codes" on public.discount_codes; create policy "admin manages codes" on public.discount_codes for all to authenticated using (true) with check (true); -- مفيش أي صلاحية للزائر على الجدول نفسه، عشان محدش يقدر يشوف كل الأكواد -- 3) دالة التحقق من كود الخصم: ترجع قيمة الكود بس، ولو الكود غلط ترجع فاضي create or replace function public.check_discount(p_code text) returns numeric language plpgsql security definer set search_path = public as $$ declare v numeric; begin select value into v from public.discount_codes where upper(code) = upper(trim(coalesce(p_code,''))) and active; return v; -- بترجع null لو الكود مش موجود أو متوقف end; $$; grant execute on function public.check_discount(text) to anon; grant execute on function public.check_discount(text) to authenticated; -- 4) تحديث دالة تسجيل الطلب: بقى ليها خانة كود الخصم -- الخصم والإجمالي النهائي بيتحسبوا جوه قاعدة البيانات نفسها (مش من المتصفح) create or replace function public.place_order( p_customer_name text, p_phone text, p_address text, p_payment text, p_items jsonb, p_subtotal numeric, p_fee numeric, p_total numeric, p_discount_code text default null ) returns bigint language plpgsql security definer set search_path = public as $$ declare new_id bigint; d numeric := 0; final_total numeric; begin if p_discount_code is not null and length(trim(p_discount_code)) > 0 then select value into d from public.discount_codes where upper(code) = upper(trim(p_discount_code)) and active; d := least(coalesce(d, 0), p_subtotal + p_fee); end if; final_total := greatest(p_subtotal + p_fee - d, 0); insert into public.orders (customer_name, phone, address, payment, items, subtotal, fee, total, discount, discount_code) values (p_customer_name, p_phone, p_address, p_payment, p_items, p_subtotal, p_fee, final_total, d, case when d > 0 then upper(trim(p_discount_code)) else null end) returning id into new_id; return new_id; end; $$; grant execute on function public.place_order(text,text,text,text,jsonb,numeric,numeric,numeric,text) to anon; -- 5) تحديث دالة متابعة الطلب: بترجع كمان بيانات المندوب والخصم create or replace function public.get_order_status(p_id bigint, p_phone text) returns json language plpgsql security definer set search_path = public as $$ declare o record; begin select id, status, customer_name, items, total, fee, created_at, driver_name, driver_phone, discount into o from public.orders where id = p_id and right(regexp_replace(coalesce(phone,''), '\D', '', 'g'), 10) = right(regexp_replace(coalesce(p_phone,''), '\D', '', 'g'), 10); if not found then return null; end if; return json_build_object( 'id', o.id, 'status', o.status, 'name', split_part(coalesce(o.customer_name,''), ' ', 1), 'items', o.items, 'total', o.total, 'fee', o.fee, 'created_at', o.created_at, 'driver_name', o.driver_name, 'driver_phone', o.driver_phone, 'discount', o.discount ); end; $$; grant execute on function public.get_order_status(bigint, text) to anon; grant execute on function public.get_order_status(bigint, text) to authenticated; -- 6) تحديث كاش قاعدة البيانات عشان الأعمدة الجديدة تظهر فوراً notify pgrst, 'reload schema';
